# 논문을 요약해 주는 ai 연구원 완성하기

논문 요약 시스템은 단순히 텍스트를 요약하는 것을 넘어, **1) PDF 텍스트 추출, 2) 텍스트 전처리 및 청킹(Chunking), 3) LLM을 통한 요약 수행**의 단계를 거치게 됩니다.

---

## 1. AI 연구원 구축 절차

1. **환경 구성 및 라이브러리 설치**: API 호출을 위한 `openai` 라이브러리와 PDF 파일 가공을 위한 `PyMuPDF` 또는 `PyPDF2` 또는 `pdfplumber`를 설치합니다.
2. **PDF 텍스트 추출 및 전처리**: PDF 파일의 텍스트를 페이지별 또는 섹션별로 읽어옵니다. 논문은 글자 수가 많으므로 API 입력 제한(Context Window)을 넘지 않도록 적절한 크기로 쪼개는 작업(청킹)이 필요합니다.
3. **요약 프롬프트 설계**: AI가 단순 번역이 아닌 연구원 관점에서 **연구 목적, 방법론, 핵심 결과, 한계점**을 도출하도록 페르소나와 구조를 지정합니다.
4. **결합 및 출력 (Map-Reduce 방식)**: 쪼개진 텍스트 블록들을 각각 요약(Map)한 뒤, 요약본들을 다시 하나로 모아 종합 요약(Reduce)을 완성합니다.

---

## 2. 파이썬 구현 코드

### 1. 앞에서 생성한 결과 텍스트 파일을 읽어 요약하는 예제 코드

>파일명 : step5.py

```python
from openai import OpenAI
import os

# 환경 변수에서 OpenAI API 키를 가져옵니다.
api_key = os.getenv('OPENAI_API_KEY')

def summarize_txt(file_path: str): # ① 텍스트 파일을 입력받아 요약하는 함수 정의
    # OpenAI 클라이언트 인스턴스를 생성합니다.
    client = OpenAI(api_key=api_key)

    # ② 지정된 경로의 텍스트 파일을 '읽기(r)' 모드로 열어 내용을 가져옵니다.
    with open(file_path, 'r', encoding='utf-8') as f:
        txt = f.read()

    # ③ LLM(대형 언어 모델)에게 전달할 시스템 프롬프트를 구성합니다.
    # 역할(요약 봇), 수행할 작업(문제 인식 및 주장 파악), 출력 포맷을 명시하고 본문을 삽입합니다.
    system_prompt = f'''
    너는 다음 글을 요약하는 봇이다. 아래 글을 읽고, 저자의 문제 인식과 주장을 파악하고, 주요 내용을 요약하라. 

    작성해야 하는 포맷은 다음과 같다. 
    
    # 제목

    ## 저자의 문제 인식 및 주장 (15문장 이내)
    
    ## 저자 소개

    
    =============== 이하 텍스트 ===============

    { txt }
    '''

    # 프롬프트 구성이 잘 되었는지 콘솔에 출력하여 확인합니다.
    print(system_prompt)
    print('=========================================')

    # ④ OpenAI API의 Chat Completions 엔드포인트를 호출하여 요약을 요청합니다.
    response = client.chat.completions.create(
        model="gpt-4.1-mini",  # 사용할 모델을 지정합니다.
        temperature=0.1,       # 답변의 일관성과 정형성을 높이기 위해 낮은 온도로 설정합니다.
        messages=[
            {"role": "system", "content": system_prompt}, # 시스템 역할을 부여하고 프롬프트를 전달합니다.
        ]
    )

    # 생성된 답변 결과 텍스트만 추출하여 반환합니다.
    return response.choices[0].message.content

# 요약할 대상 파일의 경로를 설정합니다.
file_path = 'output/과정기반 작물모형을 이용한 웹 기반 밀 재배관리 의사결정 지원시스템 설계 및 구축_with_preprocessing2.txt'

# 함수를 호출하여 요약 작업을 수행하고 결과를 변수에 저장합니다.
summary = summarize_txt(file_path)
print(summary)

# ⑤ 요약이 완료된 텍스트 내용을 새로운 파일로 저장합니다.
with open('output/crop_model_summary.txt', 'w', encoding='utf-8') as f:
    f.write(summary)

```

### 2. 결과 텍스트 파일을 읽어 청킹으로 나누어서 요약하는 예제 코드

**ChatGPT 같은 LLM 기반의 RAG(Retrieval-Augmented Generation) 시스템 구축**시 사용할 수 있게, 정제된 텍스트를 적절한 크기로 쪼개는 **청킹(Chunking)** 작업이 필수적입니다. LLM(대규모 언어 모델)이나 RAG(검색 증강 생성) 시스템을 구축할 때 텍스트를 적절한 크기로 자르는 과정을 청킹(Chunking)이라고 하며, 잘려진 텍스트 조각을 청크(Chunk)라고 합니다.

* **Map-Reduce 알고리즘**: 수십 페이지에 달하는 롱폼(Long-form) 논문을 분석할 때 발생할 수 있는 토큰 유실이나 망각 현상을 방지하기 위해 분할 분석 후 병합하는 구조를 채택했습니다.
* **시스템 프롬프트 제어**: `system` 역할을 활용해 AI에게 명확한 '연구원' 페르소나를 부여하고, 출력 구조(배경, 방법론, 결과, 결론)를 강제함으로써 결과물의 가독성과 전문성을 높였습니다.

>파일명: step6.py

```python
from openai import OpenAI
import os

# 환경 변수에서 OpenAI API 키를 가져옵니다.
api_key = os.getenv('OPENAI_API_KEY')

# 1. OpenAI 클라이언트 초기화
# 환경 변수에 OPENAI_API_KEY가 설정되어 있어야 합니다. (os.environ["OPENAI_API_KEY"])
client = OpenAI(api_key=api_key)

def read_text_file(file_path: str) -> str:
    """지정된 경로의 텍스트 파일을 읽어 내용을 반환하는 함수"""
    with open(file_path, 'r', encoding='utf-8') as f:
        txt = f.read()
    return txt


def chunk_text(text, max_chars=5000):
    """컨텍스트 한계를 고려하여 텍스트를 일정 글자 수 단위로 나누는 함수"""
    chunks = []
    current_chunk = ""
    
    # 문장이나 줄바꿈 기준으로 분할하여 단락이 깨지는 것을 방지
    for line in text.split('\n'):
        if len(current_chunk) + len(line) < max_chars:
            current_chunk += line + "\n"
        else:
            chunks.append(current_chunk.strip())
            current_chunk = line + "\n"
            
    if current_chunk:
        chunks.append(current_chunk.strip())
    return chunks

def summarize_chunk(text_chunk):
    """각 텍스트 조각을 요약하는 함수 (Map 단계)"""
    response = client.chat.completions.create(
        model="gpt-4o",  # 또는 gpt-4o-mini
        messages=[
            {
                "role": "system", 
                "content": "당신은 전문 AI 연구원입니다. 제공된 논문 텍스트 조각의 핵심 연구 내용, 방법론, 수치적 성과를 누락 없이 상세히 요약하세요."
            },
            {"role": "user", "content": f"다음 논문 내용을 요약해 주세요:\n\n{text_chunk}"}
        ],
        temperature=0.1
    )
    return response.choices[0].message.content

def final_comprehensive_summary(summary_list):
    """분할 요약된 결과물들을 하나로 모아 최종 논문 보고서 형태로 정제하는 함수 (Reduce 단계)"""
    combined_summary = "\n\n".join(summary_list)
    
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system", 
                "content": (
                    "당신은 수석 연구원입니다. 분할 요약된 논문 내용들을 종합하여 "
                    "구조화된 최종 연구 보고서를 작성하세요. "
                    "반드시 다음 구조를 포함해야 합니다:\n"
                    "1. 연구 배경 및 목적 (Introduction)\n"
                    "2. 제안 방법론 및 핵심 알고리즘 (Methodology)\n"
                    "3. 실험 결과 및 주요 성과 (Results)\n"
                    "4. 결론 및 향후 연구 방향 (Conclusion)"
                )
            },
            {"role": "user", "content": f"다음은 논문의 파트별 요약본입니다. 이를 종합하여 최종 보고서를 작성해 주세요:\n\n{combined_summary}"}
        ],
        temperature=0.3
    )
    return response.choices[0].message.content

def ai_researcher_pipeline(txt_file_path):
    """전체 논문 요약 파이프라인 실행 함수"""
    raw_text = read_text_file(txt_file_path)
    
    print("2. 텍스트 청킹 진행 중...")
    chunks = chunk_text(raw_text, max_chars=5000)
    print(f"총 {len(chunks)} 개의 텍스트 블록이 생성되었습니다.")
    
    print("3. 파트별 개별 요약 진행 중...")
    partial_summaries = []
    for i, chunk in enumerate(chunks):
        print(f"[{i+1}/{len(chunks)}] 블록 요약 중...")
        summary = summarize_chunk(chunk)
        partial_summaries.append(summary)
        
    print("4. 최종 종합 보고서 생성 중...")
    final_report = final_comprehensive_summary(partial_summaries)
    
    return final_report


# 요약할 대상 파일의 경로를 설정합니다.
file_path = 'output/과정기반 작물모형을 이용한 웹 기반 밀 재배관리 의사결정 지원시스템 설계 및 구축_with_preprocessing2.txt'

# 함수를 호출하여 요약 작업을 수행하고 결과를 변수에 저장합니다.
summary = ai_researcher_pipeline(file_path)
print(summary)

# ⑤ 요약이 완료된 텍스트 내용을 새로운 파일로 저장합니다.
with open('output/crop_model_chunk_summary.txt', 'w', encoding='utf-8') as f:
    f.write(summary)

```

### 3. 정제된 텍스트를 적절한 크기로 쪼개는 청킹(Chunking)의 문제점 해결

#### 3.1. chunk_size (청크 크기)

* **정의:** 쪼개진 텍스트 조각(Chunk) 1개당 들어갈 **최대 글자 수** (또는 토큰 수)를 의미합니다.
* **설정의 의미:** 예를 들어 `chunk_size=5000`으로 설정하면, 아무리 긴 논문이나 책이라도 하나의 조각이 최대 5000자를 넘지 않도록 텍스트를 등분하게 됩니다.
* **사용 목적 (왜 조절해야 할까?):**
* **LLM의 입력 제한 때문:** ChatGPT 같은 AI 모델들은 한 번에 입력받을 수 있는 텍스트 양(컨텍스트 윈도우)이 정해져 있습니다. 너무 크게 자르면 AI가 읽지 못합니다.
* **비용 및 속도 절감:** 청크가 너무 크면 AI가 답변을 처리하는 데 시간이 오래 걸리고 API 비용도 많이 발생합니다.
* **검색 정확도 향상:** RAG 시스템에서는 질문과 가장 관련된 '특정 단락'만 찾아내야 합니다. 청크 크기가 너무 크면 쓸데없는 정보까지 함께 검색되어 AI가 엉뚱한 답변을 할 확률이 높아집니다.

---

#### 3.2. chunk_overlap (청크 중복분)

* **정의:** 텍스트를 자를 때, 앞 청크의 끝부분과 뒤 청크의 시작부분을 **서로 겹치게(중복되게) 만드는 글자 수**입니다.
* **설정의 의미:** 위의 예시 그림처럼 `chunk_overlap=500`으로 설정하면, 첫 번째 청크의 마지막 50글자가 두 번째 청크의 맨 앞부분에 한 번 더 포함됩니다.
* **사용 목적 (왜 겹치게 해야 할까?):**
* **문맥(Context)의 단절 방지:** 컴퓨터가 글을 자를 때 5000자가 딱 채워지면 문장 중간이든 단어 중간이든 무자비하게 잘라버립니다. 만약 "핵심 원인은 A이다"라는 문장이 있었는데, "핵심 원인은"에서 첫 번째 청크가 끝나고, "A이다"부터 두 번째 청크가 시작된다면 AI는 주어와 서술어를 잃어버려 문맥을 전혀 이해할 수 없게 됩니다.
* **정보 손실 최소화:** 겹치는 구간을 두면 단어와 문장의 앞뒤 연결고리가 유지되므로, 텍스트가 잘려 나가더라도 AI가 전체적인 흐름과 인과관계를 정확하게 파악할 수 있습니다.

---

#### 요약하자면

* `chunk_size=5000`은 "AI가 한 번에 읽기 가장 편하고 정확한 정보 단위를 5000자 내외로 보겠다"는 의미입니다.
* `chunk_overlap=5000`은 "글이 뚝뚝 끊겨서 뜻이 왜곡되는 것을 막기 위해 안전장치로 앞뒤 조각을 500자씩 겹쳐두겠다"는 의미입니다.

보통 정답 파일의 성격에 따라 이 값들을 튜닝하는데, 매뉴얼처럼 팩트 중심의 짧은 글은 `overlap`을 적게 주고, 소설이나 논문처럼 서사가 이어지는 글은 `overlap` 비율을 크게 주는 것이 좋습니다.

실무에서 RAG(검색 증강 생성) 시스템이나 LLM 서비스를 개발할 때 가장 널리 쓰이는 표준적인(General) 추천 값은 다음과 같습니다.

* **`chunk_size`**: **500자 ~ 1,000자** (토큰 기준으로는 약 512 ~ 1024 토큰)
* **`chunk_overlap`**: `chunk_size`의 **10% ~ 20%** (즉, 50자 ~ 200자)

하지만 이 값은 절대적인 정답이 아니며, 처리하려는 문서의 종류(성격)와 **사용하는 LLM의 특성**에 따라 조금씩 다르게 세팅해야 가장 좋은 성능을 냅니다. 상황별 최적의 가이드라인을 정리해 드릴게요.

---

### 4. 문서 성격에 따른 추천 크기

#### ① 규정집, 매뉴얼, Q&A, 표 데이터 (짧고 명확한 정보)

* **추천 값:** `chunk_size=300~500자`, `chunk_overlap=30~50자`
* **이유:** 문장 하나하나가 독립적인 팩트(Fact)를 담고 있는 경우가 많습니다. 청크를 작게 쪼개야 질문과 관련 없는 노이즈가 섞이지 않고 정확히 원하는 규칙이나 수치만 콕 집어서 검색할 수 있습니다.

#### ② 논문, 보고서, 설명서 (문맥과 인과관계가 중요한 정보)

* **추천 값:** `chunk_size=500~800자`, `chunk_overlap=100~150자`
* **이유:** 하나의 결론을 도출하기 위해 앞부분의 실험 조건이나 전제가 같이 묶여 있어야 합니다. 따라서 문맥이 끊기지 않도록 `overlap`을 조금 넉넉하게(15~20% 수준) 주는 것이 유리합니다.

#### ③ 소설, 평전, 긴 수필 (스토리가 길게 이어지는 정보)

* **추천 값:** `chunk_size=1,000자 이상`, `chunk_overlap=200자 이상`
* **이유:** 긴 대화나 사건의 흐름을 파악해야 하므로 청크 크기 자체가 커야 LLM이 전체적인 상황을 오해하지 않고 답변할 수 있습니다.

---

### 5. 최적의 크기를 정하는 3가지 기준

1. **LLM의 '눈(컨텍스트 윈도우)'의 크기**
* 최근 나오는 4o 모델이나 Claude 등은 한 번에 읽을 수 있는 양이 매우 큽니다. 따라서 예전보다 `chunk_size`를 조금 더 크게(예: 800~1000자) 가져가도 AI가 문맥을 잘 파악합니다.

2. **한 번에 AI에게 넘겨줄 청크의 개수 ($K$값)**
* 보통 검색기(Retriever)는 질문과 유사한 청크를 3~5개 정도 뽑아서 AI에게 전달합니다.
* 만약 `chunk_size`가 1,000자인데 5개를 넘겨주면 AI는 한 번에 5,000자를 읽어야 하므로 답변 속도가 느려질 수 있습니다. 반대로 300자짜리 5개를 주면 1,500자만 읽으면 되므로 속도가 매우 빠릅니다.

3. **텍스트 언어 (영어 vs 한글)**
* 오픈소스 모델을 쓸 때 **영어보다 한글이 토큰(Token)을 더 많이 차지**합니다. 글자 수로는 500자이지만 AI가 받아들이는 토큰 수로는 한글이 더 무거울 수 있으므로, 한글 문서를 다룰 때는 `chunk_size`를 너무 무리하게 크게 잡지 않는 것이 안전합니다. (500~800자 수준 권장)

### 🛠️ 실무 팁: 가장 좋은 시작점

처음 시스템을 구축하실 때는 **`chunk_size=600`, `chunk_overlap=60`** 정도로 시작하는 것이 좋습니다. 이후 AI의 답변을 모니터링하면서 다음과 같이 튜닝합니다.

* **AI가 뚱딴지같은 소리를 섞어서 한다?** $\rightarrow$ 청크 크기가 너무 큰 것이므로 **`chunk_size`를 줄입니다.**
* **AI가 "문서에 그런 내용이 없다"거나 앞뒤 맥락을 모른다?** $\rightarrow$ 정보가 끊긴 것이므로 **`chunk_overlap`을 늘립니다.**

### 6. RecursiveCharacterTextSplitter 클래스를 사용한 청크 분할

`RecursiveCharacterTextSplitter`는 LangChain에서 제공하는 **가장 권장되는 텍스트 분할기(Text Splitter)** 중 하나입니다.

단순히 글자 수만 맞춰 무조건 자르는 방식과 달리, 문맥의 의미가 최대한 깨지지 않도록 **줄바꿈, 공백, 문장 부호 등을 기준 순서(지정된 차단기 목록)에 따라 재귀적으로 탐색하며 텍스트를 적절한 크기로 쪼개줍니다.**

---

#### 6.1. 동작 원리 이해하기

이 분할기는 기본적으로 다음과 같은 구분자 배열(`separators`)을 기본값으로 가집니다.

```python
separators = ["\n\n", "\n", " ", ""]

```

작동할 때는 이 배열의 앞순위부터 차례대로 단락을 나누려고 시도합니다.

1. 먼저 `\n\n`(문단 분리)을 기준으로 텍스트를 쪼갭니다.
2. 쪼개진 덩어리가 설정한 `chunk_size`보다 여전히 크다면, 그다음 기준인 `\n`(줄바꿈)으로 더 잘게 쪼갭니다.
3. 그래도 크다면 ` `(공백) 단위로, 최종적으로는 `""`(글자 단위)로 내려가며 크기를 맞춥니다.

이 덕분에 코드 블록이나 문단, 문장이 중간에 어설프게 잘려 나가는 현상을 최소화하고 의미적 연관성이 높은 텍스트를 하나의 청크(Chunk)로 묶을 수 있습니다.

---

#### 6.2. 핵심 파라미터 4가지

* **`chunk_size`**: 각 청크가 가질 수 있는 최대 글자 수(또는 토큰 수)입니다. (기본값: 4000)
* **`chunk_overlap`**: 인접한 청크 간에 **겹치게 만들 글자 수**입니다. 문맥이 단절되는 것을 방지하기 위해 앞 청크의 끝부분 일부를 다음 청크의 시작 부분에 포함시킵니다. (기본값: 200)
* **`length_function`**: 청크의 길이를 계산하는 함수입니다. 기본값은 파이썬 자체 내장 함수인 `len`이며, 필요에 따라 토큰 기반 계산 함수로 바꿀 수 있습니다.
* **`separators`**: 분할 기준이 되는 구분자 리스트입니다. 생략 시 위의 기본 배열이 사용됩니다.

---

#### 6.3. 기본 사용법 코드 예제

`langchain-text-splitters` 패키지를 설치한 후 사용할 수 있습니다.

```bash
pip install -qU langchain-text-splitters

```

파일명 : step7.py

```python
from langchain_text_splitters import RecursiveCharacterTextSplitter

# 1. 샘플 텍스트 준비 (긴 문서라고 가정)
sample_text = """
Large Language Models (LLMs) are incredibly powerful tools for natural language processing.
They can generate essays, write code, and summarize complex texts.

However, to use them effectively with personal documents, we need Retrieval-Augmented Generation (RAG).
RAG allows us to fetch relevant documents and inject them into the LLM's prompt context.

This process requires splitting documents into smaller, meaningful pieces called 'chunks'.
Using the right text splitter ensures that semantic meaning is preserved across boundaries.
"""

# 2. Splitter 객체 생성
text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=150,       # 하나의 청크는 최대 150자 내외
    chunk_overlap=30,     # 청크 간에 30자는 서로 겹치도록 설정
    length_function=len,  # 글자 수 기준 계산
    separators=["\n\n", "\n", " ", ""] # 분할 기준 순서
)

# 3. 텍스트 분할 실행
# 문자열을 바로 쪼갤 때는 split_text를 사용합니다.
chunks = text_splitter.split_text(sample_text)

# 4. 결과 확인
for i, chunk in enumerate(chunks):
    print(f"--- Chunk {i+1} (Length: {len(chunk)}) ---")
    print(chunk)

```

#### 6.5. 실무 팁 및 고도화

- 토큰 수 기준으로 쪼개기 (Tiktoken 활용)

영어의 경우 글자 수(`len`)와 LLM이 인식하는 토큰(Token)의 수가 다릅니다. OpenAI 모델들의 컨텍스트 제한을 더 정밀하게 맞추려면 OpenAI의 토큰 카운터 라이브러리인 `tiktoken` 기반으로 분할기를 선언하는 것이 좋습니다.

```python
# OpenAI 토큰 기준으로 동작하는 스플리터
token_splitter = RecursiveCharacterTextSplitter.from_tiktoken_encoder(
    model_name="gpt-4o",
    chunk_size=100,  # 100 글자가 아니라 '100 토큰' 기준
    chunk_overlap=20
)

```

이 방식을 사용하면 LLM API 호출 시 토큰 제한 에러(`ContextWindowExceeded`)를 완벽하게 제어할 수 있어 프로덕션 레벨의 AI 애플리케이션 개발 시 표준적으로 사용됩니다.

## 3. 최종 예제

현재 사용 중인 글자 수(Character) 기반의 임의 분할 방식은 영어/한국어의 토큰 수 차이 때문에 OpenAI API의 토큰 제한(Context Limit)을 정밀하게 제어하기 어렵습니다.

`LangChain` 라이브러리의 `RecursiveCharacterTextSplitter.from_tiktoken_encoder()`를 사용하면, **OpenAI의 토큰 수 계산 방식(tiktoken)을 기반으로 문맥(줄바꿈, 공백 등)을 최대한 보존하면서 안전하게 텍스트를 청킹**할 수 있습니다.

---

### 1. 사전 준비 (라이브러리 설치)

코드를 실행하기 전에 `langchain-text-splitters`와 `tiktoken` 패키지가 설치되어 있어야 합니다. 터미널에서 아래 명령어를 실행해 주세요.

```bash

```

---

### 1. 수정된 파이썬 전체 코드

기존의 `chunk_text` 함수를 걷어내고, LangChain의 토큰 기반 스플리터를 적용한 전체 코드입니다.

```python
from openai import OpenAI
import os
# LangChain의 토큰 기반 텍스트 스플리터 임포트
from langchain_text_splitters import RecursiveCharacterTextSplitter

# 환경 변수에서 OpenAI API 키를 가져옵니다.
api_key = os.getenv('OPENAI_API_KEY')

# 1. OpenAI 클라이언트 초기화
client = OpenAI(api_key=api_key)

def read_text_file(file_path: str) -> str:
    """지정된 경로의 텍스트 파일을 읽어 내용을 반환하는 함수"""
    with open(file_path, 'r', encoding='utf-8') as f:
        txt = f.read()
    return txt

# -------------------------------------------------------------
# [수정] LangChain의 tiktoken 기반 스플리터를 사용하는 함수
# -------------------------------------------------------------
def chunk_text_by_tokens(text: str, max_tokens: int = 3000, overlap_tokens: int = 200):
    """
    tiktoken 인코더를 사용하여 텍스트를 안전하게 토큰 단위로 나누는 함수.
    줄바꿈(\n\n, \n), 공백 등 자연스러운 경계를 우선시하여 분할합니다.
    """
    text_splitter = RecursiveCharacterTextSplitter.from_tiktoken_encoder(
        model_name="gpt-4o",      # 사용할 모델 계열의 인코더 지정 (gpt-4.1-mini, gpt-4o 등 호환)
        chunk_size=max_tokens,    # 한 청크당 최대 토큰 수
        chunk_overlap=overlap_tokens # 청크 간 겹치는 토큰 수 (문맥 유실 방지)
    )
    
    # split_text는 문자열 리스트를 반환합니다.
    chunks = text_splitter.split_text(text)
    return chunks
# -------------------------------------------------------------

def summarize_chunk(text_chunk):
    """각 텍스트 조각을 요약하는 함수 (Map 단계)"""
    response = client.chat.completions.create(
        model="gpt-4.1-mini",  # 혹은 사용하시는 gpt-4o-mini 등
        messages=[
            {
                "role": "system", 
                "content": "당신은 전문 AI 연구원입니다. 제공된 논문 텍스트 조각의 핵심 연구 내용, 방법론, 수치적 성과를 누락 없이 상세히 요약하세요."
            },
            {"role": "user", "content": f"다음 논문 내용을 요약해 주세요:\n\n{text_chunk}"}
        ],
        temperature=0.1
    )
    return response.choices[0].message.content

def final_comprehensive_summary(summary_list):
    """분할 요약된 결과물들을 하나로 모아 최종 논문 보고서 형태로 정제하는 함수 (Reduce 단계)"""
    combined_summary = "\n\n".join(summary_list)
    
    response = client.chat.completions.create(
        model="gpt-4.1-mini",
        messages=[
            {
                "role": "system", 
                "content": (
                    "당신은 수석 연구원입니다. 분할 요약된 논문 내용들을 종합하여 "
                    "구조화된 최종 연구 보고서를 작성하세요. "
                    "반드시 다음 구조를 포함해야 합니다:\n"
                    "1. 연구 배경 및 목적 (Introduction)\n"
                    "2. 제안 방법론 및 핵심 알고리즘 (Methodology)\n"
                    "3. 실험 결과 및 주요 성과 (Results)\n"
                    "4. 결론 및 향후 연구 방향 (Conclusion)"
                )
            },
            {"role": "user", "content": f"다음은 논문의 파트별 요약본입니다. 이를 종합하여 최종 보고서를 작성해 주세요:\n\n{combined_summary}"}
        ],
        temperature=0.3
    )
    return response.choices[0].message.content

def ai_researcher_pipeline(txt_file_path):
    """전체 논문 요약 파이프라인 실행 함수"""
    raw_text = read_text_file(txt_file_path)
    
    print("2. 텍스트 토큰 청킹 진행 중...")
    # [수정] 글자 수 기준(5000자)에서 토큰 수 기준(3000 토큰)으로 변경
    # OpenAI API 입력 제한을 안전하게 맞추기 위해 3000~4000 토큰 수준이 적당합니다.
    chunks = chunk_text_by_tokens(raw_text, max_tokens=3000, overlap_tokens=200)
    print(f"총 {len(chunks)} 개의 텍스트 블록이 생성되었습니다.")
    
    print("3. 파트별 개별 요약 진행 중...")
    partial_summaries = []
    for i, chunk in enumerate(chunks):
        print(f"[{i+1}/{len(chunks)}] 블록 요약 중...")
        summary = summarize_chunk(chunk)
        partial_summaries.append(summary)
        
    print("4. 최종 종합 보고서 생성 중...")
    final_report = final_comprehensive_summary(partial_summaries)
    
    return final_report


# 요약할 대상 파일의 경로를 설정합니다.
file_path = 'output/과정기반 작물모형을 이용한 웹 기반 밀 재배관리 의사결정 지원시스템 설계 및 구축_with_preprocessing2.txt'

# 함수를 호출하여 요약 작업을 수행하고 결과를 변수에 저장합니다.
summary = ai_researcher_pipeline(file_path)
print("\n--- 최종 요약 보고서 결과 ---")
print(summary)
```

### 2. 주요 변경 사항 및 팁

* **`RecursiveCharacterTextSplitter.from_tiktoken_encoder` 도입**:
단순 글자 수 분할이 아닌, OpenAI 모델들이 텍스트를 받아들이는 단위인 **토큰(Token)** 기준으로 쪼갭니다. 또한 문장이나 단락 중간이 뚝 끊기지 않도록 `\n\n`, `\n`, ` `(공백)의 우선순위를 고려하여 영리하게 분할합니다.
* **`chunk_size` 매개변수 조정**:
기존 글자 수 `5000`에서 토큰 수 `3000`으로 변경했습니다. 한국어의 경우 1글자가 1~2토큰 이상을 차지하는 경우가 많으므로, LLM의 Context Window와 요약 효율을 고려했을 때 `3000` 내외가 가장 안정적입니다.
* **`chunk_overlap` 추가**:
앞뒤 청크가 살짝 겹치도록(`200` 토큰 설정) 만들어, 쪼개진 경계선 부분의 문맥이 잘려 나가 정보가 왜곡되거나 누락되는 현상을 방지했습니다.
