# **Retriever(검색 작업)** 후 LLM에 질문하고 결과 받기

현재 작성하신 Retriever 코드를 기준으로 하면 다음 단계는

```text id="w7yikm"
질문
 ↓
Retriever 검색
 ↓
Document 반환
 ↓
Context 생성
 ↓
Prompt 생성
 ↓
LLM 호출
 ↓
답변 생성
```

입니다.

이해하기 쉽도록 OpenAI 버전 예제를 먼저 보여드리겠습니다.

## 전체 코드 (Retriever + Prompt + LLM)

>파일명 : step07.py

교육용으로 이해하기 쉽도록 **상세 주석**을 추가한 버전입니다.

```python
# Chroma Vector DB 사용
from langchain_chroma import Chroma

# Hugging Face 임베딩 모델 사용
from langchain_huggingface import HuggingFaceEmbeddings

# OpenAI LLM 사용
from langchain_openai import ChatOpenAI

import os

##############################################################################
# 환경 설정
##############################################################################

# Chroma DB 컬렉션 이름
COLLECTION_NAME = "pdf_collection"

# Vector DB 저장 위치
PERSIST_DIRECTORY = "./chroma_db"

# 문장을 벡터(숫자 배열)로 변환할 임베딩 모델
EMBEDDING_MODEL = (
    "sentence-transformers/all-MiniLM-L6-v2"
)

##############################################################################
# Embedding Model 생성
##############################################################################

def get_embedding_model():
    """
    HuggingFace 임베딩 모델 로드

    역할:
        사용자의 질문(Query)과 문서(Text)를
        벡터(Vector)로 변환한다.

    예:
        "사과 재배 방법"
            ↓
        [0.123, -0.221, 0.884, ...]

    반환:
        HuggingFaceEmbeddings 객체
    """

    return HuggingFaceEmbeddings(
        model_name=EMBEDDING_MODEL
    )

##############################################################################
# Vector Store 연결
##############################################################################

def load_vector_store():
    """
    Chroma DB 연결

    역할:
        저장되어 있는 Vector DB를 로드한다.

    반환:
        Chroma 객체
    """

    embedding_model = get_embedding_model()

    return Chroma(
        collection_name=COLLECTION_NAME,
        persist_directory=PERSIST_DIRECTORY,
        embedding_function=embedding_model
    )

##############################################################################
# Retriever 검색
##############################################################################

def search(query, k=3):
    """
    사용자 질문과 가장 유사한 문서 검색

    매개변수:
        query : 사용자 질문
        k     : 가져올 문서 수

    반환:
        Document 객체 리스트

    동작 예:
        질문:
            "농작물 재배관리 의사결정"

        검색 결과:
            Document1
            Document2
            Document3
    """

    vector_store = load_vector_store()

    return vector_store.similarity_search(
        query,
        k=k
    )

##############################################################################
# LLM 생성
##############################################################################

llm = ChatOpenAI(
    model="gpt-4.1-mini",
    api_key=os.getenv("OPENAI_API_KEY"),
    temperature=0
)

##############################################################################
# RAG 함수
##############################################################################

def ask_pdf(query):
    """
    RAG(Retrieval Augmented Generation)

    처리 순서

    1. Retriever 검색
    2. Context 생성
    3. Page 정보 수집
    4. Prompt 생성
    5. LLM 답변 생성
    """

    ####################################################
    # 1. Retriever 검색
    ####################################################

    # 사용자의 질문과 가장 유사한 문서 검색
    docs = search(query)

    ####################################################
    # 2. Context 생성
    ####################################################

    # 검색된 문서 내용을 하나의 문자열로 결합
    context = "\n\n".join(
        doc.page_content
        for doc in docs
    )

    ####################################################
    # 3. 페이지 정보 추출
    ####################################################

    # 검색된 문서의 페이지 번호 수집
    pages = sorted(
        list(
            set(
                doc.metadata.get("page")
                for doc in docs
            )
        )
    )

    # 페이지 번호를 문자열로 변환
    page_text = ", ".join(
        str(page)
        for page in pages
    )

    ####################################################
    # 4. Prompt 생성
    ####################################################

    # 검색된 문서를 Context로 삽입
    prompt = f"""
당신은 스마트농업 전문가입니다.

반드시 제공된 문서 내용만 사용하여 답변하세요.

문서:
{context}

질문:
{query}

답변 작성 규칙:

1. 문서에 있는 내용만 사용한다.
2. 문서에 없는 내용은 추측하지 않는다.
3. 핵심 내용을 요약한다.
4. 답변 마지막에 참고 페이지를 표시한다.

참고 페이지:
{page_text}
"""

    ####################################################
    # 5. LLM 호출
    ####################################################

    # GPT 모델 호출
    response = llm.invoke(prompt)

    # 답변 텍스트 반환
    return response.content

##############################################################################
# 실행
##############################################################################

# 사용자 질문
query = "농작물 재배관리 의사결정에 대해 알려줘"

# RAG 실행
answer = ask_pdf(query)

# 결과 출력
print(answer)
```

## 전체 코드

```python
# =============================================================================
# Chroma Vector DB
# =============================================================================
from langchain_chroma import Chroma

# =============================================================================
# HuggingFace Embedding
# =============================================================================
from langchain_huggingface import HuggingFaceEmbeddings

# =============================================================================
# OpenAI Chat Model
# =============================================================================
from langchain_openai import ChatOpenAI

# =============================================================================
# Prompt Template
# =============================================================================
from langchain_core.prompts import PromptTemplate

# =============================================================================
# Output Parser
# =============================================================================
from langchain_core.output_parsers import StrOutputParser

# =============================================================================
# LCEL Runnable
# =============================================================================
from langchain_core.runnables import (
    RunnablePassthrough,
    RunnableLambda
)

import os

##############################################################################
# 환경 설정
##############################################################################

# Chroma Collection 이름
COLLECTION_NAME = "pdf_collection"

# Chroma DB 저장 경로
PERSIST_DIRECTORY = "./chroma_db"

# 사용할 임베딩 모델
EMBEDDING_MODEL = (
    "sentence-transformers/all-MiniLM-L6-v2"
)

##############################################################################
# Embedding Model
##############################################################################

def get_embedding_model():
    """
    텍스트를 벡터(Vector)로 변환하는 모델

    예)

    "사과 재배"

        ↓

    [0.123, -0.451, 0.992, ...]
    """

    return HuggingFaceEmbeddings(
        model_name=EMBEDDING_MODEL
    )

##############################################################################
# Vector Store
##############################################################################

def load_vector_store():
    """
    Chroma DB 연결
    """

    embedding_model = get_embedding_model()

    return Chroma(
        collection_name=COLLECTION_NAME,
        persist_directory=PERSIST_DIRECTORY,
        embedding_function=embedding_model
    )

##############################################################################
# Vector Store 로드
##############################################################################

vector_store = load_vector_store()

##############################################################################
# Retriever 생성
##############################################################################

"""
Retriever 역할

사용자 질문(Query)과
가장 유사한 문서를 검색

상위 3개 문서 반환
"""

retriever = vector_store.as_retriever(
    search_kwargs={
        "k": 3
    }
)

##############################################################################
# RunnableLambda
##############################################################################

def format_docs_with_pages(docs):
    """
    Retriever가 반환한 Document 리스트를

    1. Context 문자열 생성
    2. Page 번호 추출
    3. Context 마지막에 Page 정보 추가

    반환 예

    문서 내용1

    문서 내용2

    문서 내용3

    참고 페이지:
    3, 5, 8
    """

    ##################################################
    # 문서 내용 합치기
    ##################################################

    context = "\n\n".join(
        doc.page_content
        for doc in docs
    )

    ##################################################
    # 페이지 번호 추출
    ##################################################

    pages = sorted(
        set(
            doc.metadata.get("page")
            for doc in docs
            if doc.metadata.get("page") is not None
        )
    )

    ##################################################
    # 페이지 번호 문자열 생성
    ##################################################

    page_text = ", ".join(
        str(page)
        for page in pages
    )

    ##################################################
    # Context + Page 정보 반환
    ##################################################

    return f"""
{context}

참고 페이지:
{page_text}
"""

##############################################################################
# RunnableLambda 생성
##############################################################################

format_docs = RunnableLambda(
    format_docs_with_pages
)

##############################################################################
# Prompt Template
##############################################################################

prompt = PromptTemplate.from_template(
"""
당신은 스마트농업 전문가입니다.

반드시 제공된 문서 내용만 사용하여 답변하세요.

문서:
{context}

질문:
{query}

답변 작성 규칙:

1. 문서 내용만 사용한다.
2. 문서에 없는 내용은 추측하지 않는다.
3. 핵심 내용을 요약한다.
4. 문서에 없는 경우
   "문서에서 찾을 수 없습니다."
   라고 답변한다.
5. 참고 페이지를 함께 표시한다.

답변:
"""
)

##############################################################################
# LLM
##############################################################################

llm = ChatOpenAI(
    model="gpt-4.1-mini",
    api_key=os.getenv("OPENAI_API_KEY"),
    temperature=0
)

##############################################################################
# Output Parser
##############################################################################

"""
AIMessage

↓

문자열(String)
"""

parser = StrOutputParser()

##############################################################################
# LCEL Chain
##############################################################################

"""
동작 순서

      사용자 질문
            │
            ▼
 ┌───────────────────────┐
 │ RunnableParallel      │
 └───────────────────────┘
      │
      ├──────────────────────────┐
      │                          │
      ▼                          ▼
retriever                 RunnablePassthrough
      │                          │
      ▼                          ▼
format_docs              질문 그대로 유지
      │
      ▼
context 생성
      │
      └──────────────┬──────────────┘
                     ▼
{
   "context": 검색문서,
   "query": 사용자질문
}
                     │
                     ▼
               PromptTemplate
                     │
                     ▼
                 ChatOpenAI
                     │
                     ▼
              StrOutputParser
                     │
                     ▼

                 최종 답변
"""

chain = (
    {
        # 병렬 실행
        "context": retriever | format_docs,
        "query": RunnablePassthrough()
    }
    # 순차 실행
    | prompt
    | llm
    | parser
)

##############################################################################
# 실행
##############################################################################

query = "농작물 재배관리 의사결정에 대해 알려줘"

answer = chain.invoke(query)

print(answer)
```

# RunnableLambda를 사용하는 이유

>day12 1.14.md 파일을 Runnable, RunnableLambda을 참조하세요

예전 방식

```python
def format_docs(docs):
    return "\n\n".join(
        doc.page_content
        for doc in docs
    )
```

LCEL 체인에서 사용하려면

```python
retriever | RunnableLambda(format_docs)
```

또는

```python
RunnableLambda(
    lambda docs:
        "\n\n".join(
            doc.page_content
            for doc in docs
        )
)
```

처럼 Runnable 객체로 감싸야 합니다.

즉,

```text
일반 함수
    ↓

RunnableLambda

    ↓

LCEL에서 연결 가능
```

# 실무에서 더 많이 사용하는 형태

실제 프로젝트에서는 `RunnableLambda` 안에서 페이지 정보까지 같이 넣습니다.

```python
format_docs = RunnableLambda(
    lambda docs:
        "\n\n".join(
            f"[페이지:{doc.metadata.get('page')}]\n"
            f"{doc.page_content}"
            for doc in docs
        )
)
```

결과

```text
[페이지:5]
농작물 재배관리...

[페이지:8]
의사결정 지원 시스템...
```

이렇게 하면 LLM이 출처 페이지를 훨씬 정확하게 인용할 수 있어서 RAG 품질이 좋아집니다.
