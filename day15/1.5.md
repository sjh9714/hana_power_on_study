# 챗봇에 대한 구조 refactoring

day12/step27.py 파일을 참조하여 리트리버 기능을 구현 하기 전에 **구조 개선(Refactoring)** 챗봇에 대한 구조를 개선 하겠습니다.

현재 한 파일에 너무 많은 책임이 들어가 있습니다. 이렇게 한 개의 파일에 많은 책임(일)을 하는 경우 유지 보수 및 새로운 기능을 추가할 때 어디를 어떻게 수정을 해야하는지 어렵게 됩니다. 이를 위해 리펙토링을 진행하겠습니다.

현재 파일은 다음 7개의 역할을 동시에 수행합니다.

```text
1. Prompt 관리
2. LLM 관리
3. Session Memory 관리
4. Chain 생성
5. Chat Business Logic
6. Gradio UI
7. FastAPI 실행
```

실무에서는 보통 아래처럼 분리합니다.

# 2. llms/llm_factory.py

현재 MODEL_CONFIG + get_llm() 이동

```python
import os

from langchain_openai import ChatOpenAI
from langchain_google_genai import ChatGoogleGenerativeAI

MODEL_CONFIG = {

    "GPT-4.1 Mini": {
        "provider": "openai",
        "model": "gpt-4.1-mini"
    },

    "Gemini 2.5 Flash": {
        "provider": "gemini",
        "model": "gemini-2.5-flash"
    }
}

def get_llm(selected_model):

    config = MODEL_CONFIG[selected_model]

    if config["provider"] == "openai":

        return ChatOpenAI(
            model=config["model"],
            api_key=os.getenv(
                "OPENAI_API_KEY"
            ),
            streaming=True
        )

    return ChatGoogleGenerativeAI(
        model=config["model"],
        google_api_key=os.getenv(
            "GOOGLE_API_KEY"
        ),
        streaming=True
    )
```

# 3. memory/session_memory.py

현재 store 관련 분리

```python
from langchain_core.chat_history import (
    InMemoryChatMessageHistory
)

store = {}

def get_session_history(session_id):

    if session_id not in store:

        store[session_id] = (
            InMemoryChatMessageHistory()
        )

    return store[session_id]


def clear_session(session_id):

    if session_id in store:
        del store[session_id]
```

# 4. chains/chat_chain.py

현재 create_chain() 이동

```python
from prompts.finance_prompt import prompt

from llms.llm_factory import (
    get_llm
)

from memory.session_memory import (
    get_session_history
)

from langchain_core.output_parsers import (
    StrOutputParser
)

from langchain_core.runnables.history import (
    RunnableWithMessageHistory
)

def create_chain(selected_model):

    llm = get_llm(selected_model)

    chain = (
        prompt
        | llm
        | StrOutputParser()
    )

    return RunnableWithMessageHistory(
        chain,
        get_session_history,
        input_messages_key="question",
        history_messages_key="chat_history"
    )
```

# 5. services/chat_service.py

현재 chat() 함수 이동

```python
from chains.chat_chain import (
    create_chain
)

def chat(
    message,
    history,
    selected_model,
    session_id
):

    if history is None:
        history = []

    chain = create_chain(
        selected_model
    )

    history.append(
        {
            "role": "user",
            "content": message
        }
    )

    history.append(
        {
            "role": "assistant",
            "content": ""
        }
    )

    partial = ""

    for chunk in chain.stream(
        {
            "question": message
        },
        config={
            "configurable": {
                "session_id": session_id
            }
        }
    ):

        partial += chunk

        history[-1]["content"] = partial

        yield history, ""
```
